cmake_minimum_required(VERSION 3.20)
project(regional_water_tests LANGUAGES CXX)
enable_testing()
add_executable(regional_water_tests RegionalWaterTests.cpp)
target_compile_features(regional_water_tests PRIVATE cxx_std_20)
target_include_directories(regional_water_tests PRIVATE ../../src)
add_test(NAME regional_water_model COMMAND regional_water_tests)
add_executable(regional_water_multi_tests MultiRegionalWaterTests.cpp)
target_compile_features(regional_water_multi_tests PRIVATE cxx_std_20)
target_include_directories(regional_water_multi_tests PRIVATE ../../src)
add_test(NAME regional_water_multi_model COMMAND regional_water_multi_tests)
add_executable(regional_water_polygon_tests PolygonRegionalWaterTests.cpp)
target_compile_features(regional_water_polygon_tests PRIVATE cxx_std_20)
target_include_directories(regional_water_polygon_tests PRIVATE ../../src)
add_test(NAME regional_water_polygon_model COMMAND regional_water_polygon_tests)
